## Integration Workflow Examples

### Orchestrator Integration Pattern

```python
async def execute_architecture_review_workflow(planning_results):
    """Complete workflow example for orchestrator integration."""

    # 1. Prepare architecture review input
    review_input = {
        "task_id": f"arch-review-{generate_timestamp()}",
        "operation_type": "architecture_review",
        "review_scope": {
            "specification_file": planning_results["source_spec"],
            "plan_files": planning_results["generated_plans"],
            "maturity_stage": get_project_maturity_stage()
        },
        "quality_requirements": get_quality_requirements_for_stage(),
        "research_requirements": {
            "context7_research": True,
            "token_budget": 5000
        }
    }

    # 2. Execute architecture review
    try:
        review_result = await execute_agent("architecture-agent", review_input)

        # 3. Process results based on status
        if review_result["status"] == "SUCCESS":
            assessment = review_result["operation_result"]["overall_assessment"]

            if assessment["approval_status"] == "approved":
                # Proceed to task generation
                return {
                    "decision": "proceed_to_tasks",
                    "validated_plans": review_result["operation_result"]["plans_reviewed"],
                    "quality_gate_passed": True
                }

            elif assessment["approval_status"] == "approved_with_conditions":
                # Present to human for review
                return {
                    "decision": "human_review_required",
                    "review_results": review_result,
                    "conditions": extract_high_priority_recommendations(review_result)
                }

            else:  # rejected
                return {
                    "decision": "return_to_planning",
                    "feedback": extract_critical_feedback(review_result)
                }

        elif review_result["status"] == "NEEDS_CLARIFICATION":
            # Handle clarification requests
            return {
                "decision": "enhance_plans",
                "enhancement_requests": review_result["proposed_next_steps"]
            }

        else:  # ERROR
            # Handle errors with appropriate escalation
            return {
                "decision": "error_escalation",
                "error_context": review_result
            }

    except Exception as e:
        # Handle unexpected errors
        return {
            "decision": "error_escalation",
            "error": f"Architecture review failed: {str(e)}"
        }
```
